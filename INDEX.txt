<script>
  let finalScriptUrl = ""; 

  // Format Tanggal Indonesia (Asli Anda)
  const formatTanggalIndo = (dateString) => {
    if (!dateString) return "-";
    const bulanIndo = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
    const d = new Date(dateString);
    return `${d.getDate()} ${bulanIndo[d.getMonth()]} ${d.getFullYear()}`;
  };

  const toBase64 = file => new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = () => resolve({ base64: reader.result.split(',')[1], mimeType: file.type, name: file.name });
    reader.onerror = error => reject(error);
  });

  // Logika Tampil/Sembunyi Pindahan (Request Baru)
  document.getElementsByName('statusPindahan').forEach(radio => {
    radio.addEventListener('change', function() {
      const formPindahan = document.getElementById('formPindahan');
      formPindahan.style.display = this.value === 'YA' ? 'block' : 'none';
      document.getElementById('kelasTujuan').required = (this.value === 'YA');
      document.getElementById('fotoSuratPindah').required = (this.value === 'YA');
    });
  });

  document.getElementById('psbForm').addEventListener('submit', async function(e) {
    e.preventDefault();
    const btn = document.getElementById('submitBtn');
    const loading = document.getElementById('loadingOverlay');
    const formCont = document.getElementById('formContainer');
    
    btn.style.display = 'none';
    loading.style.display = 'block';

    try {
      const isPindahan = document.querySelector('input[name="statusPindahan"]:checked').value;
      
      const payload = {
        namaLengkap: document.getElementById('namaLengkap').value,
        tempatLahir: document.getElementById('tempatLahir').value,
        tanggalLahir: document.getElementById('tanggalLahir').value,
        alamatSantri: document.getElementById('alamatSantri').value,
        asalSekolah: document.getElementById('asalSekolah').value,
        alamatSekolah: document.getElementById('alamatSekolah').value,
        tahunLulus: document.getElementById('tahunLulus').value,
        namaIbu: document.getElementById('namaIbu').value,
        noTelp: document.getElementById('noTelp').value,
        // Payload Pindahan
        isPindahan: isPindahan,
        jenjang: isPindahan === "YA" ? document.querySelector('input[name="jenjang"]:checked')?.value : "",
        kelas: isPindahan === "YA" ? document.getElementById('kelasTujuan').value : "",
        // Berkas
        fotoKK: await toBase64(document.getElementById('fotoKK').files[0]),
        fotoKTP: await toBase64(document.getElementById('fotoKTP').files[0]),
        fotoDiri: await toBase64(document.getElementById('fotoDiri').files[0]),
        fotoIjazah: await toBase64(document.getElementById('fotoIjazah').files[0]),
        fotoSuratPindah: (isPindahan === "YA" && document.getElementById('fotoSuratPindah').files[0]) 
                         ? await toBase64(document.getElementById('fotoSuratPindah').files[0]) : null
      };

      google.script.run
        .withSuccessHandler(function(res) {
          if(res.status === "Sukses") {
            finalScriptUrl = res.scriptUrl;
            // Isi data rincian (Asli Anda)
            document.getElementById('resNama').innerText = res.data.nama;
            document.getElementById('resTempat').innerText = res.data.tempatLahir;
            document.getElementById('resTanggal').innerText = formatTanggalIndo(res.data.tanggalLahir);
            document.getElementById('resAlamat').innerText = res.data.alamat;
            document.getElementById('resSekolah').innerText = res.data.asalSekolah;
            document.getElementById('resAlamatSekolah').innerText = res.data.alamatSekolah;
            document.getElementById('resTahunLulus').innerText = res.data.tahunLulus;
            document.getElementById('resIbu').innerText = res.data.namaIbu;
            document.getElementById('resWa').innerText = res.data.wa;

            // Link Berkas (Asli Anda)
            document.getElementById('lnkKK').href = res.data.links.kk;
            document.getElementById('lnkKTP').href = res.data.links.ktp;
            document.getElementById('lnkFoto').href = res.data.links.foto;
            document.getElementById('lnkIjazah').href = res.data.links.ijazah;

            // Tampilkan Detail Pindahan jika ada
            if(isPindahan === "YA") {
              document.getElementById('resPindahanBox').style.display = 'block';
              document.getElementById('resPindahanDetail').innerText = payload.jenjang + " / " + payload.kelas;
              if(res.data.links.suratPindah) {
                document.getElementById('colSuratPindah').style.display = 'block';
                document.getElementById('lnkSuratPindah').href = res.data.links.suratPindah;
              }
            }

            loading.style.display = 'none';
            formCont.style.display = 'none';
            document.getElementById('successPage').style.display = 'block';
          } else {
            alert("Gagal: " + res.message);
            btn.style.display = 'block'; loading.style.display = 'none';
          }
        })
        .withFailureHandler(function(err) {
          alert("Kesalahan: " + err);
          btn.style.display = 'block'; loading.style.display = 'none';
        })
        .processForm(payload);
    } catch (err) {
      alert("Error: " + err);
      btn.style.display = 'block'; loading.style.display = 'none';
    }
  });

  document.getElementById('btnRelog').addEventListener('click', function() {
    finalScriptUrl ? window.top.location.href = finalScriptUrl : location.reload();
  });
</script>
